function populateQuizData(quiz) {

  if (quiz.settings) {

  if (db) {
    db.collection("quizzes").doc(quizId).get().then(doc => {

        populateQuizData(doc.data());

    }).catch(() => {});
  }

$('save-btn').onclick = () => {
  const title = $('qtitle').value.trim();
  const desc = $('qdesc').value.trim();
  if (!title) {
    $('status-msg').textContent = 'Please enter a quiz title.';
    $('status-msg').style.color = 'var(--bad)';
    return;
  }
  settingsState.tabViolationPenalty = parseInt($('pen-val').value, 10) || 0;
  settingsState.timePerQuestion = parseInt($('time-val').value, 10) || 20;

  const cards = $('questions-container').querySelectorAll('.q-card');
  const questions = [];
  cards.forEach(c => {
    const q = c.querySelector('.q-text').value.trim();
    const correct = c.querySelector('.q-corr').value.trim();
    const w1 = c.querySelector('.q-w1').value.trim();
    const w2 = c.querySelector('.q-w2').value.trim();
    const w3 = c.querySelector('.q-w3').value.trim();
    const exp = c.querySelector('.q-exp').value.trim();
    if (q || correct) {
      questions.push({
        q,
        correct,
        wrongs: [w1, w2, w3].filter(Boolean),
        explanation: exp
      });
    }
  });

  if (questions.length === 0) {
    $('status-msg').textContent = 'Please add at least one question.';
    $('status-msg').style.color = 'var(--bad)';
    return;
  }

  $('status-msg').textContent = 'Saving to Firebase...';
  $('status-msg').style.color = 'var(--mu)';
  if (db) {
    db.collection("quizzes").doc(quizId).set({
      title,
      description: desc,
      settings: settingsState,
      questions,
      updatedAt: Date.now()
    }).then(() => {
      $('status-msg').textContent = 'Saved Successfully to Student Quiz File! Quiz is live.';
      $('status-msg').style.color = 'var(--ok)';
    }).catch(err => {
      $('status-msg').textContent = 'Error saving: ' + err.message;
      $('status-msg').style.color = 'var(--bad)';
    });
  } else {
    $('status-msg').textContent = 'Firebase is not initialized.';
    $('status-msg').style.color = 'var(--bad)';
  }
};
