function shuffle(arr) {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

function initDisplay(data) {
  quizData = data;
  $('title').textContent = data.title || "Quiz";

  if (st.enableTimer === false) {

  } else {
    $('r-timer').textContent = "Each question has " + (st.timePerQuestion || 20) + " seconds. When time runs out, it moves on automatically.";
  }
  if (st.enableTabViolation === false) {
    $('r-tab').style.display = 'none';

    $('r-tab').textContent = "Every tab switch or minimizing costs -" + (st.tabViolationPenalty != null ? st.tabViolationPenalty : 1) + " point(s).";
